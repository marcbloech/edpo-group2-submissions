// ════════════════════════════════════════
// CONNECTIONS
// ════════════════════════════════════════

// User entry: three lifecycle endpoints, all on the signup service
User > Signup Service: POST /api/signup
User > Signup Service: POST /api/upgrade
User > Signup Service: POST /api/deactivate

// Step 1: Signup publishes one of three lifecycle events to Kafka
Signup Service > Kafka: SignupRequestedEvent [style: dashed, color: blue]
Signup Service > Kafka: UpgradeRequestedEvent [style: dashed, color: blue]
Signup Service > Kafka: AccountDeactivationRequestedEvent [style: dashed, color: blue]

// Step 2: Process Service's MessageListener routes each event type
// to the matching BPMN process (Kafka→Zeebe ingress bridge)
Kafka > Process Service: SignupRequested → signup-process [style: dashed, color: blue]
Kafka > Process Service: UpgradeRequested → upgrade-process [style: dashed, color: blue]
Kafka > Process Service: AccountDeactivationRequested → deactivation-process [style: dashed, color: blue]

// Step 3: Zeebe issues commands to workers (orchestration)
Zeebe > Signup Service: validate-signup, activate-account, apply-tier-upgrade, deactivate-account [color: red, label: "commands (gRPC)"]
Zeebe > Payment Service: process-payment, refund-payment [color: red, label: "commands (gRPC)"]
Zeebe > Notification Service: send-notification [color: red, label: "command (gRPC)"]
Zeebe > Slack: postMessage (payment-process) [color: red, label: "Marketplace connector"]
Process Service > Zeebe: start process instance (3 process IDs) [color: red]

// Step 4: Workers publish events to Kafka after job completion (choreography)
Payment Service > Kafka: PaymentReceived/Failed/RefundedEvent [style: dashed, color: blue]
Notification Service > Kafka: NotificationSent/FailedEvent [style: dashed, color: blue]
Signup Service > Kafka: AccountActivatedEvent [style: dashed, color: blue]

// Scanners publish events to Kafka
Market Scanner > Kafka: MarketAlertEvent [style: dashed, color: blue]
Bluesky Scanner > Kafka: SocialTrendEvent [style: dashed, color: blue]

// Notification Service consumes scanner events from Kafka (choreography)
Kafka > Notification Service: MarketAlertEvent, SocialTrendEvent [style: dashed, color: blue]

// Monitoring
Zeebe > Operate: process data [color: grey]
